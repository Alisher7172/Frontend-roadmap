Create this portfolio using React

Quick steps — Windows PowerShell

npm create vite@latest my-portfolio -- --template react
cd my-portfolio
npm install
npm run dev

Copy these into src/. Optionally copy styles from `CSS/day48.css` into your React `src/index.css`.

src/App.jsx (click to expand) 
import React from 'react'
import ProjectsGrid from './ProjectsGrid'
import './index.css'

export default function App(){
  return (
    <main className="container">
      <section className="about-section">
        <h2>My React Portfolio</h2>
        <p>Built with React + Vite</p>
      </section>
      <ProjectsGrid />
    </main>
  )
}
src/ProjectsGrid.jsx 
import React from 'react'
import ProjectCard from './ProjectCard'

const projects = [
  { title: 'React Landing', desc: 'Animated landing built in React', thumb: '/assets/project-1.png' },
  { title: 'Hooks Timer', desc: 'Timers and effects with useEffect', thumb: '/assets/project-2.png' },
  { title: 'Forms & Filters', desc: 'Controlled forms and filtering', thumb: '/assets/project-3.png' }
]

export default function ProjectsGrid(){
  return (
    <section className="projects-section">
      <h3>Featured Projects</h3>
      <div className="projects-grid">
        {projects.map(p => (<ProjectCard key={p.title} {...p} />))}
      </div>
    </section>
  )
}
src/ProjectCard.jsx 
import React from 'react'

export default function ProjectCard({ title, desc, thumb }){
  return (
    <article className="project-card">
      <img src={thumb} alt={title} />
      <h4>{title}</h4>
      <p>{desc}</p>
      <div className="project-links">
        <a href="#">Live</a>
        <a href="#">Source</a>
      </div>
    </article>
  )
}
